Complete Backend SQL Schema (Tables, RLS & RPCs)

Paste this migration into your Supabase SQL Editor (or run supabase_migration.sql) to deploy all tables, RLS policies, triggers, and RPC procedures: 

-- 1. Profiles Table & Helpers
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  name TEXT NOT NULL DEFAULT 'Hunter',
  level INT NOT NULL DEFAULT 1 CHECK (level BETWEEN 1 AND 100),
  exp INT NOT NULL DEFAULT 0 CHECK (exp >= 0),
  total_exp INT NOT NULL DEFAULT 0 CHECK (total_exp >= 0),
  streak INT NOT NULL DEFAULT 0 CHECK (streak >= 0),
  last_active_day DATE,
  sound_enabled BOOLEAN NOT NULL DEFAULT true,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. Quests
CREATE TABLE IF NOT EXISTS public.quests (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  title TEXT NOT NULL,
  notes TEXT,
  category TEXT NOT NULL DEFAULT 'General',
  exp INT NOT NULL DEFAULT 25,
  done BOOLEAN NOT NULL DEFAULT false,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  completed_at TIMESTAMPTZ
);

-- 3. Solved Challenges
CREATE TABLE IF NOT EXISTS public.solved_challenges (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  challenge_id TEXT NOT NULL,
  solved_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_user_challenge UNIQUE (user_id, challenge_id)
);

-- 4. Daily EXP Heatmap
CREATE TABLE IF NOT EXISTS public.daily_exp (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  day DATE NOT NULL,
  exp_earned INT NOT NULL DEFAULT 0,
  CONSTRAINT uq_user_day UNIQUE (user_id, day)
);

-- 5. Activity Log (trimmed to 30 rows)
CREATE TABLE IF NOT EXISTS public.activity_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  text TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 6. Unlocked Achievements
CREATE TABLE IF NOT EXISTS public.unlocked_achievements (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  achievement_id TEXT NOT NULL,
  unlocked_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT uq_user_achievement UNIQUE (user_id, achievement_id)
);

XELI Terminal Security — Help

XELI Hunter Terminal Security Help Topics:

• User Name: Identifies your Hunter avatar on quests, daily streaks, and the terminal leaderboard.
• Security Passkey: Verifies your clearance on this device. Any custom passcode can be entered.
• Guild / Realm: Determines your division alignment (e.g., Shadow Monarch, Hunter Association, White Tiger) displayed on your hunter license.
• Switching Accounts: You may log off or change callsigns anytime via the Log On / Log Off header button or by clicking your mini-badge in the left navigation panel.
